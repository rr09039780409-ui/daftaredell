    70→/* ═══════════════════════════════════════════════════════════════
    71→   Text-wrapping utility (handles Persian / RTL text)
    72→   ═══════════════════════════════════════════════════════════════ */
    73→
    74→function wrapTextToLines(
    75→  ctx: CanvasRenderingContext2D,
    76→  text: string,
    77→  maxWidth: number
    78→): string[] {
    79→  const paragraphs = text.split(/\n/);
    80→  const lines: string[] = [];
    81→
    82→  for (const paragraph of paragraphs) {
    83→    if (!paragraph.trim()) {
    84→      lines.push("");
    85→      continue;
    86→    }
    87→
    88→    const words = paragraph.split(/\s+/);
    89→    let currentLine = "";
    90→
    91→    for (const word of words) {
    92→      if (!word) continue;
    93→
    94→      const candidate = currentLine ? `${currentLine} ${word}` : word;
    95→      const measured = ctx.measureText(candidate);
    96→
    97→      if (measured.width > maxWidth && currentLine) {
    98→        lines.push(currentLine);
    99→        currentLine = word;
   100→      } else {
   101→        currentLine = candidate;
   102→      }
   103→    }
   104→
   105→    if (currentLine) {
   106→      lines.push(currentLine);
   107→    }
   108→  }
   109→
   110→  return lines;
   111→}

   113→/* ═══════════════════════════════════════════════════════════════
   114→   Canvas rendering engine
   115→   ═══════════════════════════════════════════════════════════════ */
   116→
   117→function renderCanvas(
   118→  canvas: HTMLCanvasElement,
   119→  content: string,
   120→  settings: ReaderSettings,
   121→  containerWidth: number
   122→): void {
   123→  const ctx = canvas.getContext("2d");
   124→  if (!ctx) return;
   125→
   126→  const dpr = window.devicePixelRatio || 1;
   127→  const { fontSize, theme, fontFamily, lineHeight } = settings;
   128→  const colors = getThemeColors(theme);
   129→
   130→  /* Build font string with fallbacks */
   131→  const fontStr = `${fontSize}px "${fontFamily}", "Vazirmatn", Tahoma, "Segoe UI", sans-serif`;
   132→
   133→  /* Reset transform before measuring */
   134→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   135→  ctx.font = fontStr;
   136→
   137→  /* Word-wrap */
   138→  const maxWidth = Math.max(containerWidth - CANVAS_PADDING_X * 2, 60);
   139→  const lines = wrapTextToLines(ctx, content, maxWidth);
   140→
   141→  const lineSpacing = fontSize * lineHeight;
   142→  const totalHeight =
   143→    lines.length * lineSpacing + CANVAS_PADDING_Y * 2;
   144→
   145→  /* Size canvas (physical × dpr, CSS at logical) */
   146→  canvas.width = containerWidth * dpr;
   147→  canvas.height = totalHeight * dpr;
   148→  canvas.style.width = `${containerWidth}px`;
   149→  canvas.style.height = `${totalHeight}px`;
   150→
   151→  /* Scale context for crisp rendering on HiDPI */
   152→  ctx.scale(dpr, dpr);
   153→
   154→  /* Background fill */
   155→  ctx.fillStyle = colors.bg;
   156→  ctx.fillRect(0, 0, containerWidth, totalHeight);
   157→
   158→  /* Text settings */
   159→  ctx.font = fontStr;
   160→  ctx.fillStyle = colors.text;
   161→  ctx.direction = "rtl";
   162→  ctx.textAlign = "right";
   163→  ctx.textBaseline = "top";
   164→
   165→  /* Draw each line */
   166→  const x = containerWidth - CANVAS_PADDING_X;
   167→  for (let i = 0; i < lines.length; i++) {
   168→    if (lines[i]) {
   169→      ctx.fillText(lines[i], x, CANVAS_PADDING_Y + i * lineSpacing);
   170→    }
   171→  }
   172→}

   198→
   199→  /* ─── refs ─── */

   201→  const wrapperRef = useRef<HTMLDivElement>(null);
   202→
   203→  /* ─── derived ─── */
   204→  const colors = useMemo(
   205→    () => getThemeColors(readerSettings.theme),
   206→    [readerSettings.theme]
   207→  );
   208→
   209→  const currentThemeMeta = useMemo(
   210→    () => THEME_OPTIONS.find((t) => t.value === readerSettings.theme)!,
   211→    [readerSettings.theme]
   212→  );
   213→
   214→  const ThemeIcon = currentThemeMeta.icon;
   215→
   216→  const inBookResults = useMemo(() => {

   222→      const idx = lines[i].toLowerCase().indexOf(q);

   224→        results.push({ lineIndex: i, text: lines[i], matchStart: idx, matchEnd: idx + q.length });
   225→      }
   226→    }
   227→    return results;
   228→  }, [readerSearchQuery, bookContent]);
   229→
   230→  useEffect(() => { setReaderSearchResults(inBookResults); }, [inBookResults]);

   239→    setReaderSearchResults([]);

   241→
   242→  const handleReaderResultClick = useCallback((lineIndex: number) => {

   244→    setReaderSearchOpen(false);
   245→    setReaderSearchQuery("");
   246→    setReaderSearchResults([]);
   247→  }, [setScrollToLine]);
   248→
   249→  /* ═══════════════════════════════════════════════════════════
   250→     Font loading
   251→     ═══════════════════════════════════════════════════════════ */
   252→
   253→  const [fontReady, setFontReady] = useState(true);
   254→
   255→  useEffect(() => {
   256→    let cancelled = false;
   257→    document.fonts
   258→      .load(`20px "${readerSettings.fontFamily}"`)
   259→      .then(() => { if (!cancelled) setFontReady(true); })

   261→    return () => { cancelled = true; };
   262→  }, [readerSettings.fontFamily]);
   263→

   269→    const canvas = canvasRef.current;
   270→    const wrapper = wrapperRef.current;

   273→  }, [bookContent, readerSettings, fontReady]);
   274→

   276→  useEffect(() => {
   277→    paint();
   278→  }, [paint]);
   279→
   280→  /* Re-paint on container resize */
   281→  useEffect(() => {
   282→    const wrapper = wrapperRef.current;
   283→    if (!wrapper) return;
   284→
   285→    const ro = new ResizeObserver(() => paint());

   287→    return () => ro.disconnect();

   307→        const lineSpacing = readerSettings.fontSize * readerSettings.lineHeight;

   311→        return;
   312→      }

   315→      if (saved) {
   316→        const pos = parseFloat(saved);
   317→        wrapperRef.current?.scrollTo({ top: pos });
   318→      }
   319→    }, 200);
   320→    return () => clearTimeout(timer);
   321→  }, [selectedBook?.id, bookContent, scrollToLine, readerSettings.fontSize, readerSettings.lineHeight, setScrollToLine]);
   322→
   323→  const handleScroll = useCallback(() => {
   324→    const el = wrapperRef.current;
   325→    if (!el) return;
   326→    const max = el.scrollHeight - el.clientHeight;
   327→    setScrollPercent(max > 0 ? Math.round((el.scrollTop / max) * 100) : 0);

   329→      localStorage.setItem(`bookmark-${selectedBook.id}`, String(el.scrollTop));
   330→    }
   331→  }, [selectedBook]);
   332→

   342→

   347→  }, [readerSettings.fontSize, updateReaderSettings]);
   348→
   349→  const decreaseFont = useCallback(() => {

   351→      fontSize: Math.max(12, readerSettings.fontSize - 1),

   353→  }, [readerSettings.fontSize, updateReaderSettings]);
   354→
   355→  /* ═══════════════════════════════════════════════════════════
   356→     Anti-copy: prevent context menu & text selection on canvas
   357→     ═══════════════════════════════════════════════════════════ */
   358→
   359→  const handleCanvasContextMenu = useCallback(
   360→    (e: React.MouseEvent) => e.preventDefault(),
   361→    []
   362→  );
   363→
   364→  /* ═══════════════════════════════════════════════════════════
   365→     Empty state
   366→     ═══════════════════════════════════════════════════════════ */
   367→
   368→  if (!bookContent) {
   369→    return (
   370→      <motion.div
   371→        initial={{ opacity: 0, y: 12 }}
   372→        animate={{ opacity: 1, y: 0 }}
   373→        transition={{ duration: 0.4 }}
   374→        dir="rtl"
   375→        className="flex flex-1 flex-col items-center justify-center gap-4 p-8"
   376→      >
   377→        <div
   378→          className="flex h-24 w-24 items-center justify-center rounded-full"
   379→          style={{ backgroundColor: `${colors.text}10` }}
   380→        >
   381→          <FileText className="size-12" style={{ color: `${colors.text}60` }} />
   382→        </div>
   383→        <p className="text-lg font-semibold" style={{ color: colors.text }}>
   384→          محتوایی برای نمایش وجود ندارد
   385→        </p>
   386→        <p className="text-sm" style={{ color: `${colors.text}70` }}>
   387→          لطفاً کتابی را از کتابخانه انتخاب کنید
   388→        </p>
   389→      </motion.div>
   390→    );
   391→  }
   392→
   393→  /* ═══════════════════════════════════════════════════════════
   394→     Render
   395→     ═══════════════════════════════════════════════════════════ */
   396→
   397→  return (
   398→    <div dir="rtl" className="relative flex flex-1 flex-col overflow-hidden">
   399→      <AnimatePresence>
   400→        {readerSearchOpen && (
   401→          <motion.div
   402→            initial={{ opacity: 0, height: 0 }}
   403→            animate={{ opacity: 1, height: "auto" }}
   404→            exit={{ opacity: 0, height: 0 }}
   405→            transition={{ duration: 0.2 }}
   406→            className="overflow-hidden border-b"
   407→            style={{ borderColor: `${colors.text}12`, backgroundColor: `${colors.bg}f5` }}
   408→          >
   409→            <div className="mx-auto flex max-w-3xl flex-col gap-2 px-4 py-3">
   410→              <div className="flex items-center gap-2">
   411→                <Search className="size-4 shrink-0" style={{ color: `${colors.text}60` }} />
   412→                <input
   413→                  ref={readerSearchInputRef}
   414→                  type="search"
   415→                  value={readerSearchQuery}
   416→                  onChange={(e) => setReaderSearchQuery(e.target.value)}
   417→                  placeholder="جستجو در متن این کتاب..."
   418→                  dir="rtl"
   419→                  autoFocus
   420→                  className="h-8 flex-1 rounded-md border-0 bg-transparent px-2 text-sm outline-none placeholder:text-muted-foreground/50"
   421→                  style={{ color: colors.text }}

   424→                  <span className="shrink-0 text-xs tabular-nums font-medium" style={{ color: `${colors.text}60` }}>
   425→                    {readerSearchResults.length} نتیجه

   494→      {/* ── Bottom toolbar ── */}

   503→        }}
